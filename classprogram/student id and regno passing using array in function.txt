#include <iostream>
using namespace std;

void studentget(int studentid[],int array[])//formal argument{
    for(int i=0;i<10;i++){
        cin>>studentid[i]>>array[i];
    }
}
void studentdisplay(int studentid[],int array[])//formal argument{
    for(int i=0;i<10;i++){
        cout<<studentid[i]<<array[i]<<endl;
    }
}
int main(){
    int studentid[10];
    int regno[10];
    studentget(studentid,regno)//actual argument;
    studentdisplay(studentid,regno)//actual argument;
}
